cmake_minimum_required(VERSION 3.10)
project(EvoRef LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 99)
enable_testing()
find_package(Glog REQUIRED)
find_package(gflags REQUIRED)
find_package(GTest REQUIRED)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

include_directories(${CMAKE_SOURCE_DIR})
add_subdirectory(chunk)
add_subdirectory(feature)
add_subdirectory(utils)
add_subdirectory(index)
add_subdirectory(storage)
add_subdirectory(3party/xdelta3)
add_subdirectory(encoder)
add_library(delta_compression STATIC delta_compression.cpp)
target_link_libraries(delta_compression chunker feature index storage utils xdelta3 glog gflags)
add_executable(delta main.cpp recover_file.cpp)
target_link_libraries(delta delta_compression stdc++fs)
add_executable(delta_policy_test delta_policy_test.cpp)
target_link_libraries(delta_policy_test GTest::gtest_main)
add_test(NAME delta_policy_test COMMAND delta_policy_test)
